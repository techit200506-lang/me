from flask import Flask, request, render_template_string from cryptography.fernet import Fernet app = Flask(__name__) # Clé de chiffrement key = Fernet.generate_key() cipher = Fernet(key) HTML = """ Cryptage des données

🔐 Cryptage des données

Petit projet Python + HTML

ℹ️ Fonctionnement

Le texte est envoyé à Python. Python chiffre ou déchiffre les données puis affiche le résultat.

""" @app.route("/", methods=["GET", "POST"]) def home(): result = "" if request.method == "POST": message = request.form.get("message", "") operation = request.form.get("operation", "") try: if operation == "encrypt": # Chiffrement encrypted = cipher.encrypt( message.encode("utf-8") ) result = encrypted.decode("utf-8") elif operation == "decrypt": # Déchiffrement decrypted = cipher.decrypt( message.encode("utf-8") ) result = decrypted.decode("utf-8") except Exception: result = "❌ Erreur : texte chiffré invalide." return render_template_string( HTML, result=result ) if __name__ == "__main__": app.run( debug=True )
